   LCD pin 12 (D5) -> Arduino digital pin 4 
   LCD pin 13 (D6) -> Arduino digital pin 3

  lcd.begin(16, 2); 
  // start writing on the first row and first column. 
  lcd.setCursor(0, 0); 
  // Print a message to the LCD. 
  lcd.print("hello, world!"); 
}
